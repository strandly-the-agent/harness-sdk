use std::sync::Arc;
use std::sync::atomic::{AtomicUsize, Ordering};
use std::time::{Duration, Instant};

use strands::models::bedrock::{BedrockModel, BedrockModelConfig, BedrockModelOptions};
use strands::{Agent, CancelOptions};
use tokio::io::{AsyncReadExt, AsyncWriteExt};
use tokio::net::{TcpListener, TcpStream};

async fn read_request(socket: &mut TcpStream) {
    let mut buffer = vec![0u8; 65536];
    let mut seen = Vec::new();
    loop {
        let Ok(n) = socket.read(&mut buffer).await else { return };
        if n == 0 { return; }
        seen.extend_from_slice(&buffer[..n]);
        if let Some(head_end) = seen.windows(4).position(|w| w == b"\r\n\r\n") {
            let head = String::from_utf8_lossy(&seen[..head_end]).to_ascii_lowercase();
            let length: usize = head.lines().find_map(|l| l.strip_prefix("content-length:")).map_or(0, |v| v.trim().parse().unwrap_or(0));
            if seen.len() >= head_end + 4 + length { return; }
        }
    }
}

fn response(status: &str, error_type: Option<&str>, body: &str) -> String {
    let header = error_type.map_or(String::new(), |kind| format!("x-amzn-errortype: {kind}\r\n"));
    format!("HTTP/1.1 {status}\r\ncontent-type: application/json\r\n{header}content-length: {}\r\n\r\n{body}", body.len())
}

async fn serve(scenario: &'static str) -> (String, Arc<AtomicUsize>) {
    let listener = TcpListener::bind("127.0.0.1:0").await.unwrap();
    let url = format!("http://{}", listener.local_addr().unwrap());
    let hits = Arc::new(AtomicUsize::new(0));
    let counter = Arc::clone(&hits);
    tokio::spawn(async move {
        loop {
            let Ok((mut socket, _)) = listener.accept().await else { return };
            let counter = Arc::clone(&counter);
            tokio::spawn(async move {
                loop {
                    read_request(&mut socket).await;
                    counter.fetch_add(1, Ordering::SeqCst);
                    let reply = match scenario {
                        "ok" => response("200 OK", None, r#"{"output":{"message":{"role":"assistant","content":[{"text":"Hi"}]}},"stopReason":"end_turn","usage":{"inputTokens":1,"outputTokens":1,"totalTokens":2},"metrics":{"latencyMs":5}}"#),
                        "throttled" => response("429 Too Many Requests", Some("ThrottlingException"), r#"{"message":"Too many requests"}"#),
                        "overflow" => response("400 Bad Request", Some("ValidationException"), r#"{"message":"Input is too long for requested model."}"#),
                        "html_500" => "HTTP/1.1 500 Internal Server Error\r\ncontent-type: text/html\r\ncontent-length: 11\r\n\r\n<html>bad</".to_owned(),
                        "garbage_200" => response("200 OK", None, r#"{"output":"#),
                        "empty_200" => response("200 OK", None, r#"{}"#),
                        _ => {
                            tokio::time::sleep(Duration::from_secs(60)).await;
                            return;
                        }
                    };
                    if socket.write_all(reply.as_bytes()).await.is_err() { return; }
                }
            });
        }
    });
    (url, hits)
}

async fn agent_for(url: String) -> Agent {
    let sdk_config = aws_config::SdkConfig::builder()
        .endpoint_url(url)
        .behavior_version(aws_config::BehaviorVersion::latest())
        .build();
    let model = BedrockModel::new(BedrockModelOptions {
        config: BedrockModelConfig { stream: Some(false), ..Default::default() },
        region: Some("us-west-2".to_owned()),
        client_config: Some(sdk_config),
        api_key: Some("k".to_owned()),
        ..Default::default()
    });
    Agent::builder().model(model).no_retry_strategy().printer(false).build().unwrap()
}

async fn run(scenario: &'static str) {
    let (url, hits) = serve(scenario).await;
    let mut agent = agent_for(url).await;
    let started = Instant::now();
    let outcome = tokio::time::timeout(Duration::from_secs(40), agent.invoke("hi")).await;
    let text = match outcome {
        Err(_) => "HUNG (40 s)".to_owned(),
        Ok(Ok(result)) => format!("ok stop={:?} text={:?}", result.stop_reason(), result.to_string()),
        Ok(Err(error)) => format!("err {error:?}"),
    };
    eprintln!("{scenario}: {:.2}s requests={} -> {}", started.elapsed().as_secs_f32(), hits.load(Ordering::SeqCst), text.chars().take(330).collect::<String>());
}

#[tokio::test(flavor = "multi_thread", worker_threads = 2)]
async fn scenarios() {
    for scenario in ["ok", "throttled", "overflow", "html_500", "garbage_200", "empty_200"] {
        run(scenario).await;
    }
}

#[tokio::test(flavor = "multi_thread", worker_threads = 2)]
async fn cancel_while_the_server_is_silent() {
    let (url, hits) = serve("silent").await;
    let mut agent = agent_for(url).await;
    let handle = agent.cancel_handle();
    tokio::spawn(async move {
        tokio::time::sleep(Duration::from_millis(500)).await;
        handle.cancel(CancelOptions::default());
    });
    let started = Instant::now();
    let outcome = tokio::time::timeout(Duration::from_secs(10), agent.invoke("hi")).await;
    eprintln!("silent+cancel: {:.2}s requests={} -> {:?}", started.elapsed().as_secs_f32(), hits.load(Ordering::SeqCst), outcome.map(|r| r.map(|ok| ok.stop_reason()).map_err(|e| e.to_string())).map_err(|_| "HUNG (10 s)"));
}

#[tokio::test(flavor = "multi_thread", worker_threads = 2)]
async fn error_messages() {
    let (url, _hits) = serve("html_500").await;
    let mut agent = agent_for(url).await;
    let error = agent.invoke("hi").await.unwrap_err();
    eprintln!("html_500 display ({} chars): {error}", error.to_string().len());
    let mut agent = agent_for("http://127.0.0.1:9".to_owned()).await;
    let error = agent.invoke("hi").await.unwrap_err();
    eprintln!("refused display ({} chars): {error}", error.to_string().len());
}
